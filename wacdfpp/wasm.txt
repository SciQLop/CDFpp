[binaries]
c = 'emcc'
cpp = 'em++'
ar = 'emar'
exe_wrapper = 'node'
# xsimd, for the SIMD time conversions, is configured with CMake.
cmake = 'cmake'

[built-in options]
# -msimd128: WebAssembly SIMD, in every browser since 2023 (Safari 16.4). The compiler
# vectorizes byte swaps, deflate and other loops with it: 10 to 35% faster on several paths.
c_args = ['-fwasm-exceptions', '-msimd128']
c_link_args = []
cpp_args = ['-fwasm-exceptions', '-msimd128']
cpp_link_args = []


[properties]
# Emscripten always needs an exe wrapper. Again,
# maybe Meson could just know this.
needs_exe_wrapper = true

[host_machine]
system = 'wasm'
cpu_family = 'wasm'
cpu = 'wasm'
endian = 'little'
